  // 安全：sandbox 不带 allow-same-origin，浏览器拦截跨同源访问；srcdoc 内联，无网络请求
  // 高度自适应：iframe 内桥脚本通过 postMessage 汇报 documentElement.scrollHeight
  // 大块降级：>64KB 不渲染，加提示，原 pre 保留供 Prism 高亮显示源码
  const HTML_BLOCK_MAX_BYTES = 65536;
  const HTML_BLOCK_BRIDGE = '<script>(function(){const o=new ResizeObserver(()=>{parent.postMessage({type:"rt-html-resize",height:document.documentElement.scrollHeight},"*")});o.observe(document.documentElement)})();</script>';

  function _renderHtmlCodeBlocks(wrapper) {
    wrapper.querySelectorAll('pre code.language-html').forEach(code => {
      const html = code.textContent || '';
      const pre = code.parentElement;
      if (!pre || pre.tagName !== 'PRE') return;

      if (html.length > HTML_BLOCK_MAX_BYTES) {
        const note = document.createElement('div');
        note.className = 'rt-html-too-large';
        note.textContent = '⚠ HTML 块过大（' + (html.length/1024).toFixed(1) + 'KB > 64KB），已折叠不渲染';
        pre.parentElement.insertBefore(note, pre);
        return;
      }

      const iframe = document.createElement('iframe');
      iframe.className = 'rt-html-block';
      iframe.setAttribute('sandbox', 'allow-scripts');
      iframe.style.cssText = 'width:100%;border:0;min-height:120px;height:0;background:transparent';
      iframe.srcdoc = HTML_BLOCK_BRIDGE + html;

      pre.parentElement.replaceChild(iframe, pre);
    });
  }
